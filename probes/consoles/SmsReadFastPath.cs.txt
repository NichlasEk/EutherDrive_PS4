        // Sega cartridge ROM only. Mapper writes remain authoritative: read the
        // live bank registers on every access; cartridge RAM and other mappers
        // continue through the original implementation below.
        private int _ps4SmsRomSize;
        private uint _ps4SmsRomMask, _ps4SmsBankMask;
        private bool _ps4SmsPowerOfTwo;
        private byte ReadSmsMemory(ushort a)
        {
            if (a < 0xC000 && md_main.g_masterSystemMapper == SmsMapperType.Sega &&
                _smsCartridgeEnabled && !_smsSegaRamEnabled && md_main.g_masterSystemRom != null)
            {
                int size = md_main.g_masterSystemRomSize;
                if (size != _ps4SmsRomSize)
                {
                    _ps4SmsRomSize = size;
                    _ps4SmsPowerOfTwo = size > 0 && (size & (size - 1)) == 0;
                    _ps4SmsRomMask = (uint)(size - 1);
                    _ps4SmsBankMask = (uint)(Math.Max(1, size >> 14) - 1);
                }
                if (_ps4SmsPowerOfTwo)
                {
                    bool fixedWindow = a < 0x0400;
                    uint bank = a < 0x4000 ? _smsBank0 : (a < 0x8000 ? _smsBank1 : _smsBank2);
                    uint index = fixedWindow ? (uint)a & _ps4SmsRomMask :
                        (((bank & _ps4SmsBankMask) << 14) + (uint)(a & 0x3FFF)) & _ps4SmsRomMask;
                    byte value = md_main.g_masterSystemRom[index];
                    TrackLastRead(a, value, !fixedWindow, index);
                    LogSmsRomReadIfNeeded(a, value, index, !fixedWindow, fixedWindow ? "fixed" : null);
                    return value;
                }
            }
            return ReadSmsMemoryOriginal(a);
        }
